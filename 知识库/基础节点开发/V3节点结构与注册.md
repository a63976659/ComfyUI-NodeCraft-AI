## V3 节点最小化模板

```python
from comfy_api.latest import ComfyExtension, io
from typing_extensions import override

class MyNode(io.ComfyNode):
    @classmethod
    def define_schema(cls):
        return io.Schema(
            node_id="MyNode",
            display_name="My Custom Node",
            category="my_category",
            inputs=[
                io.Image.Input("image"),
                io.Float.Input("strength", default=1.0, min=0.0, max=1.0, step=0.01),
            ],
            outputs=[
                io.Image.Output("IMAGE"),
            ],
        )

    @classmethod
    def execute(cls, image, strength):
        result = image * strength
        return io.NodeOutput(result)

class MyExtension(ComfyExtension):
    @override
    async def get_node_list(self) -> list[type[io.ComfyNode]]:
        return [MyNode]

async def comfy_entrypoint() -> MyExtension:
    return MyExtension()
```

## io.Schema 完整字段

| 字段 | 类型 | 必需 | 说明 |
|-----|------|------|------|
| node_id | str | ✓ | 全局唯一 ID，建议前缀化：Author_NodeName |
| display_name | str | | UI 显示名称 |
| category | str | | 菜单路径，如 image/transform |
| description | str | | 节点工具提示 |
| inputs | list | ✓ | 输入对象列表 |
| outputs | list | ✓ | 输出对象列表 |
| hidden | list | | 隐藏输入（服务器提供） |
| is_output_node | bool | | 标记为输出节点 |
| is_experimental | bool | | 标记为实验功能 |
| is_deprecated | bool | | 标记为已废弃 |
| not_idempotent | bool | | 禁用缓存 |
| accept_all_inputs | bool | | 接受任意类型输入 |
| search_aliases | list | | 搜索别名列表 |

## node_id 最佳实践

- 使用全局唯一前缀：`AuthorName_NodeName`
- 永不改变 node_id（会破坏已保存工作流）
- 用 display_name 修改用户可见名称
- 用 search_aliases 增加可发现性

## 完整生命周期示例

```python
class FullLifecycleNode(io.ComfyNode):
    @classmethod
    def define_schema(cls):
        return io.Schema(
            node_id="FullLifecycleNode",
            display_name="Full Lifecycle Demo",
            category="example",
            inputs=[
                io.Image.Input("image"),
                io.Float.Input("threshold", default=0.5, min=0.0, max=1.0),
                io.Image.Input("optional_ref", optional=True, lazy=True),
            ],
            outputs=[
                io.Image.Output("IMAGE"),
                io.Mask.Output("MASK"),
            ],
            hidden=[io.Hidden.unique_id],
        )

    @classmethod
    def validate_inputs(cls, image, threshold, optional_ref=None):
        if threshold == 0.0:
            return "Threshold cannot be exactly 0"
        return True

    @classmethod
    def fingerprint_inputs(cls, image, threshold, optional_ref=None):
        return threshold

    @classmethod
    def check_lazy_status(cls, image, threshold, optional_ref=None):
        if threshold > 0.8 and optional_ref is None:
            return ["optional_ref"]
        return []

    @classmethod
    def execute(cls, image, threshold, optional_ref=None):
        from comfy_api.latest import ComfyAPISync
        node_id = cls.hidden.unique_id
        api = ComfyAPISync()
        
        gray = (image[:,:,:,0] * 0.299 + image[:,:,:,1] * 0.587 + image[:,:,:,2] * 0.114)
        mask = (gray > threshold).float()
        api.execution.set_progress(1, 2)
        
        result = image * mask.unsqueeze(-1)
        if optional_ref is not None:
            result = result + optional_ref * (1 - mask.unsqueeze(-1))
        
        api.execution.set_progress(2, 2)
        return io.NodeOutput(result, mask)
```

## 异步执行

```python
class AsyncNode(io.ComfyNode):
    @classmethod
    async def execute(cls, url):
        import aiohttp
        async with aiohttp.ClientSession() as session:
            async with session.get(url) as response:
                text = await response.text()
        return io.NodeOutput(text)
```

## 条件执行分支

```python
@classmethod
def execute(cls, allow, image):
    if not allow:
        return io.NodeOutput(block_execution="Gate is closed")
    return io.NodeOutput(image)
```

## 进度报告

```python
from comfy_api.latest import ComfyAPISync

@classmethod
def execute(cls, steps):
    api = ComfyAPISync()
    for i in range(steps):
        # 处理逻辑...
        api.execution.set_progress(i + 1, steps)
    return io.NodeOutput("done")
```
