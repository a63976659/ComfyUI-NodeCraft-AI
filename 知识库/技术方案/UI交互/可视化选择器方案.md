# 可视化选择器方案

## 来源
参考工具/项目：ComfyUI-Model-Visual-Loader（https://github.com/a63976659/ComfyUI-Model-Visual-Loader）
定位：ComfyUI 节点内嵌的可视化选择器 UI，将下拉列表升级为带缩略图、分类、搜索的网格选择界面

## 适用场景
- ComfyUI 节点需要从大量选项中选择的场景（模型、图片、预设等）
- 需要缩略图/预览辅助选择的 widget
- 需要搜索、分类筛选的列表型 widget
- 需要多选/堆叠选择的复合型 widget

## 核心思路
通过 `app.registerExtension()` 在节点注册阶段劫持 `onNodeCreated` 生命周期，注入自定义 DOM Widget 替代原有下拉框，以 localStorage 维护节点独立状态，通过 widget callback 劫持解决 ComfyUI 多时机值设置导致的状态冲突。

## 实现步骤

### 1. 前端扩展注册

```javascript
// 加载样式表
const link = document.createElement("link");
link.rel = "stylesheet";
link.href = new URL(`./style.css?v=${Date.now()}`, import.meta.url).href;
document.head.appendChild(link);

// 注册扩展
app.registerExtension({
    name: "Comfy.VisualModelLoader",
    async beforeRegisterNodeDef(nodeType, nodeData, app) {
        const TARGET_NODES = {
            "Checkpoint加载器": "checkpoint",
            "Lora加载器": "lora",
            "Lora堆加载器": "lora_stack",
        };
        
        if (TARGET_NODES[nodeData.name]) {
            const modelType = TARGET_NODES[nodeData.name];
            const onNodeCreated = nodeType.prototype.onNodeCreated;
            
            nodeType.prototype.onNodeCreated = function () {
                const r = onNodeCreated?.apply(this, arguments);
                
                // 节点独立状态容器
                if (!this.visualContext) {
                    this.visualContext = { category: "全部", search: "" };
                }
                
                // 创建并添加 DOM Widget
                const domWidget = createSingleWidget(this, modelType, topOffset, this.visualContext);
                this.addDOMWidget(modelType + "_selector", "visual_list", domWidget.widget, {
                    getValue() { return ""; },
                    setValue(v) { }
                });
                
                return r;
            };
        }
    }
});
```

**关键点**：
- `beforeRegisterNodeDef` 阶段拦截，在节点定义注册前注入
- 每个节点实例有独立的 `visualContext`，多节点互不干扰
- `addDOMWidget` 将完整 DOM 元素融入 ComfyUI 节点系统

### 2. 状态管理（StateManager）

```javascript
class StateManager {
    constructor(nodeId, modelType, savedContext) {
        this.prefix = `ComfyVL_${modelType}_${nodeId}_`;
        this.savedContext = savedContext;
    }
    
    getInitialCategory() {
        // 三层优先级：localStorage > savedContext > 默认值
        const val = localStorage.getItem(this.prefix + "Cat");
        if (val) return val;
        if (this.savedContext?.category) return this.savedContext.category;
        return "全部";
    }
    
    saveCategory(val) { localStorage.setItem(this.prefix + "Cat", val); }
    saveSelection(val) { localStorage.setItem(this.prefix + "Sel", val); }
    getLastSelection() { return localStorage.getItem(this.prefix + "Sel") || ""; }
    saveScroll(val) { localStorage.setItem(this.prefix + "Scr", String(val)); }
    
    restoreScroll(gridElement) {
        const target = parseInt(localStorage.getItem(this.prefix + "Scr") || "0");
        // 多次尝试恢复（防止异步加载竞态）
        [0, 100, 300, 600].forEach(t => setTimeout(() => {
            if (gridElement.scrollHeight > gridElement.clientHeight) {
                gridElement.scrollTop = target;
            }
        }, t));
    }
}
```

### 3. Widget 值同步（真理值模式）

解决 ComfyUI 在初始化/加载工作流/粘贴节点时多次设置 widget 值导致的状态混乱：

```javascript
function createSingleWidget(node, modelType, topPadding, savedContext) {
    // 从 localStorage 读取真理值
    let TRUTH_VALUE = state.getLastSelection();
    let isInitializing = true;
    let isUserSelecting = false;
    
    // 劫持 widget callback
    const w = node.widgets[0];
    const origin = w.callback;
    w.callback = function(v) {
        // 用户主动选择：允许更新
        if (isUserSelecting) {
            state.saveSelection(v);
            TRUTH_VALUE = v;
            syncSelection(v);
            if (origin) origin.apply(this, arguments);
            return;
        }
        
        // 初始化阶段冲突值：拦截
        if (isInitializing && TRUTH_VALUE && normalizePath(v) !== normalizePath(TRUTH_VALUE)) {
            w.value = TRUTH_VALUE;  // 强制保持真理值
            return;
        }
        
        // 正常透传
        state.saveSelection(v);
        TRUTH_VALUE = v;
        syncSelection(v);
        if (origin) origin.apply(this, arguments);
    };
    
    // 数据加载后强制应用真理值
    VisualAPI.getModels(modelType).then(data => {
        if (TRUTH_VALUE) {
            syncSelection(TRUTH_VALUE);
            node.widgets[0].value = TRUTH_VALUE;
        }
        renderGrid();
        setTimeout(() => { isInitializing = false; }, 500);
    });
}
```

### 4. 堆叠选择（多选模式）

```javascript
function createStackWidget(node, modelType, topPadding, savedContext) {
    let stackData = JSON.parse(configWidget?.value || "[]");
    let isInternalUpdate = false;
    
    function saveStack(refreshUI = true) {
        const str = JSON.stringify(stackData);
        // 加锁防止回调触发重绘
        isInternalUpdate = true;
        if (w.callback) w.callback(str);
        isInternalUpdate = false;
        if (refreshUI) updateStackUI();
    }
    
    function addToStack(name) {
        stackData.push({ name, strength_model: 1.0, strength_clip: 1.0 });
        saveStack(true);
        // 滚到底部
        setTimeout(() => { stackContainer.scrollTop = stackContainer.scrollHeight; }, 0);
    }
    
    // 回调劫持：内部更新不重绘，外部更新全量重绘
    w.callback = function(v) {
        if (isInternalUpdate) {
            if (origin) origin.apply(this, arguments);
            return;
        }
        try {
            const newData = JSON.parse(v);
            if (Array.isArray(newData)) {
                stackData = newData;
                updateStackUI();
                renderGrid();
            }
        } catch {}
        if (origin) origin.apply(this, arguments);
    };
}
```

### 6. 前后端通信

```javascript
const VisualAPI = {
    async getModels(type) {
        if (type.includes("stack")) type = "lora";  // 类型转换
        const res = await api.fetchApi(`/visual_loader/models?type=${type}`);
        return await res.json();
    },
    async getNote(type, name) {
        const res = await api.fetchApi(`/visual_loader/notes?type=${type}&name=${encodeURIComponent(name)}`);
        return (await res.json()).content || "";
    },
    async saveNote(type, name, content) {
        await api.fetchApi("/visual_loader/notes", {
            method: "POST",
            body: JSON.stringify({ type, name, content })
        });
    }
};
```

## ComfyUI 适配要点

1. **生命周期劫持**：通过 `beforeRegisterNodeDef` + `nodeType.prototype.onNodeCreated` 注入 UI
2. **DOM Widget 整合**：使用 `this.addDOMWidget()` 让自定义 UI 成为节点的一部分
3. **Widget 同步**：劫持 `widget.callback` 确保自定义 UI 和 ComfyUI widget 系统双向同步
4. **onConfigure 恢复**：加载工作流时在 `onConfigure` 中触发 callback 恢复状态
5. **CSS 作用域**：所有选择器使用 `.vl-` 前缀，避免与 ComfyUI 样式冲突
6. **样式表加载**：通过 `import.meta.url` 相对路径 + 时间戳防缓存

## 注意事项

1. **Widget 值冲突**：ComfyUI 在多个时机（初始化、加载、粘贴、undo/redo）设置 widget 值，必须用"真理值 + 初始化锁"模式防御
2. **堆叠内部更新**：修改参数时需加锁（`isInternalUpdate`），否则 callback 触发全量重绘破坏输入体验
3. **滚动恢复竞态**：异步加载数据后 DOM 尚未完全渲染，需要多次延迟尝试恢复滚动位置
4. **Hover 边缘闪烁**：使用 `scale()` 从中心放大替代 `translateY()`，避免鼠标在卡片边缘反复触发
5. **路径规范化**：Windows/Linux 路径分隔符不同，比较路径时需 `normalizePath()` 统一为 `/`
6. **localStorage 隔离**：键名用 `节点类型_节点ID_` 前缀，确保同类型多节点互不干扰
