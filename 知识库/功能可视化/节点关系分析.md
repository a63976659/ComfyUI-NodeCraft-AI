# 节点关系与功能可视化分析

## 概述
功能可视化帮助开发者理解插件的节点结构、数据流和依赖关系，便于调试和优化。

## 节点关系图

### 节点类型分类
ComfyUI 中的节点按功能可分为：
- **输入节点**：加载图片、模型、文本等外部资源
- **处理节点**：对输入数据进行变换（如图像增强、模型推理）
- **输出节点**：将结果保存到文件或显示在界面
- **控制节点**：条件分支、循环、延迟等流程控制

### 节点连接分析
```python
# 分析节点的输入输出连接
def analyze_connections(node):
    """分析节点的连接关系"""
    inputs = {}
    outputs = {}
    
    # INPUT_TYPES 定义的输入
    input_types = node.INPUT_TYPES()
    if "required" in input_types:
        for name, spec in input_types["required"].items():
            inputs[name] = {
                "type": spec[0] if isinstance(spec, (list, tuple)) else spec,
                "required": True
            }
    
    if "optional" in input_types:
        for name, spec in input_types["optional"].items():
            inputs[name] = {
                "type": spec[0] if isinstance(spec, (list, tuple)) else spec,
                "required": False
            }
    
    # RETURN_TYPES 定义的输出
    return_types = node.RETURN_TYPES
    for i, rtype in enumerate(return_types):
        outputs[f"output_{i}"] = {"type": rtype}
    
    return inputs, outputs
```

## 数据流可视化

### 张量形状追踪
```python
def trace_tensor_shapes(node, input_tensors):
    """追踪张量在节点中的形状变化"""
    print(f"[{node.__class__.__name__}] 输入:")
    for name, tensor in input_tensors.items():
        if hasattr(tensor, 'shape'):
            print(f"  {name}: {list(tensor.shape)}")
    
    # 执行节点获取输出
    outputs = node.execute(**input_tensors)
    
    print(f"[{node.__class__.__name__}] 输出:")
    if isinstance(outputs, (list, tuple)):
        for i, out in enumerate(outputs):
            if hasattr(out, 'shape'):
                print(f"  output_{i}: {list(out.shape)}")
    
    return outputs
```

## 插件结构可视化

### 生成节点清单
```python
def generate_node_manifest(node_class_mapping):
    """生成插件的节点清单（可用于可视化）"""
    manifest = {
        "nodes": [],
        "categories": set()
    }
    
    for display_name, node_class in node_class_mapping.items():
        node_info = {
            "name": display_name,
            "class": node_class.__name__,
            "category": getattr(node_class, "CATEGORY", "unknown"),
            "inputs": list(node_class.INPUT_TYPES().get("required", {}).keys()),
            "outputs": list(node_class.RETURN_TYPES) if hasattr(node_class, "RETURN_TYPES") else [],
            "description": getattr(node_class, "__doc__", "")
        }
        manifest["nodes"].append(node_info)
        manifest["categories"].add(node_info["category"])
    
    manifest["categories"] = list(manifest["categories"])
    return manifest
```

## 性能热点可视化

### 节点执行时间热力图
```python
import time
from collections import defaultdict

class NodeProfiler:
    """节点执行时间分析器"""
    
    def __init__(self):
        self.times = defaultdict(list)
    
    def profile(self, node_name):
        """装饰器：记录节点执行时间"""
        def decorator(func):
            def wrapper(*args, **kwargs):
                start = time.time()
                result = func(*args, **kwargs)
                elapsed = time.time() - start
                self.times[node_name].append(elapsed)
                return result
            return wrapper
        return decorator
    
    def report(self):
        """生成性能报告"""
        for name, times in self.times.items():
            avg_time = sum(times) / len(times)
            max_time = max(times)
            print(f"[性能] {name}: 平均 {avg_time:.3f}s, 最大 {max_time:.3f}s, 调用次数 {len(times)}")
```

## 依赖关系可视化

### 插件依赖分析
```python
def analyze_dependencies(node_class):
    """分析节点的依赖关系"""
    dependencies = {
        "models": [],
        "other_nodes": [],
        "external_libs": []
    }
    
    # 从 INPUT_TYPES 分析依赖
    input_types = node_class.INPUT_TYPES()
    for name, spec in input_types.get("required", {}).items():
        input_type = spec[0] if isinstance(spec, (list, tuple)) else spec
        
        # 模型类型输入
        if input_type in ("MODEL", "CLIP", "VAE", "CONDITIONING"):
            dependencies["models"].append(input_type)
        
        # 其他节点输出类型
        elif input_type in ("LATENT", "IMAGE", "MASK"):
            dependencies["other_nodes"].append(input_type)
    
    return dependencies
```
